// TEMPORARY audit probe (not to be committed): measures the open "needs validation" leads against a real Postgres.
import type { Server } from 'node:http';
import type { AddressInfo } from 'node:net';
import { randomBytes } from 'node:crypto';
import { Kysely, PostgresDialect, sql } from 'kysely';
import pg from 'pg';
import { afterAll, beforeAll, describe, it } from 'vitest';
import { createApp } from './app.ts';
import type { DB } from './db/types.ts';
import { startTestDatabase } from './test/postgres.ts';

let stop: () => Promise<void>; let appUrl: string; let ownerDb: Kysely<DB>;
const mk = (max: number) => new Kysely<DB>({ dialect: new PostgresDialect({ pool: new pg.Pool({ connectionString: appUrl, max, types: { getTypeParser: ((oid: number, f?: any) => oid === 1082 ? (v: string) => v : pg.types.getTypeParser(oid as never, f)) as any } }) }) });
beforeAll(async () => { const t = await startTestDatabase(); stop = t.stop; appUrl = t.appUrl; ownerDb = t.db; }, 180_000);
afterAll(async () => { await stop?.(); });
const serve = async (db: Kysely<DB>, limits: object = {}) => { const s: Server = createApp({ db, limits: limits as any }).listen(0); await new Promise(d => s.once('listening', d)); return { s, base: `http://127.0.0.1:${(s.address() as AddressInfo).port}` }; };
const H = (u: string) => ({ 'content-type': 'application/json', 'x-dev-user': u, 'x-client-version': 'test-1' });
const size = async (t: string) => Number((await sql<{ n: string }>`select pg_total_relation_size(${t}::regclass)::text as n`.execute(ownerDb)).rows[0]!.n);
const cnt = async (t: string) => Number((await sql<{ n: string }>`select count(*)::text as n from ${sql.table(t)}`.execute(ownerDb)).rows[0]!.n);

describe('probe', () => {
  it('P1 storage growth from one account', async () => {
    const db = mk(10); const { s, base } = await serve(db, { commands: { max: 100000, windowMs: 60000 } });
    const before = { rh: await size('row_history'), rw: await size('refused_writes'), log: await size('log_entries') };
    const big = () => { const o: Record<string, string> = {}; for (let i = 0; i < 700; i++) o['k' + i] = randomBytes(50).toString('base64').slice(0, 64); return o; };
    let version: number | null = null; let okSaves = 0; let bytes = 0;
    const t0 = Date.now();
    for (let i = 0; i < 100; i++) {
      const body = JSON.stringify({ clientId: 'config', baseVersion: version, input: { config: { rxOverride: big() } } });
      bytes += body.length;
      const r = await fetch(`${base}/api/commands/save-config`, { method: 'POST', headers: H('probe'), body });
      const j: any = await r.json(); if (r.status === 200 || r.status === 201) { okSaves++; version = j.data?.version ?? j.version ?? (version ?? 0) + 1; } else if (i < 2) console.log('save-config status', r.status, JSON.stringify(j).slice(0, 200));
    }
    const mid = { rh: await size('row_history'), rw: await size('refused_writes') };
    let refused = 0;
    for (let i = 0; i < 100; i++) {
      const pad = randomBytes(60000).toString('base64'); // ~80 KB of junk in a bad envelope
      const r = await fetch(`${base}/api/commands/log-session`, { method: 'POST', headers: H('probe'), body: JSON.stringify({ clientId: 42, baseVersion: 'x', input: { pad } }) });
      if (r.status >= 400 && r.status < 500) refused++;
    }
    const after = { rh: await size('row_history'), rw: await size('refused_writes') };
    console.log('P1', JSON.stringify({ okSaves, requestBytes: bytes, saveSeconds: (Date.now() - t0) / 1000, rowHistoryRows: await cnt('row_history'), refusedWritesRows: await cnt('refused_writes'), refused, rowHistoryBytes: { before: before.rh, afterSaves: mid.rh, afterRefusals: after.rh }, refusedWritesBytes: { before: before.rw, afterSaves: mid.rw, afterRefusals: after.rw } }));
    s.close(); await db.destroy();
  }, 180_000);

  it('P2 import-legacy hold time and pool starvation', async () => {
    const good = (i: number) => ({ name: 'log-session', clientId: 'i' + i, input: { exerciseId: 'squat', entry: { d: '2026-10-05', ph: 'strength', w: 135, s: 3, r: 5 } } });
    const cmds = (n: number, badLast: boolean) => { const c: object[] = []; for (let i = 0; i < n; i++) c.push(good(i)); if (badLast) c.push({ name: 'log-session', clientId: 'bad', input: { exerciseId: 'squat', entry: { d: '2026-10-05', ph: 'strength', w: -5, s: 3, r: 5 } } }); return c; };
    const db = mk(2); const { s, base } = await serve(db);
    const post = (u: string, n: number, badLast: boolean) => fetch(`${base}/api/commands/import-legacy`, { method: 'POST', headers: H(u), body: JSON.stringify({ clientId: 'import-1', baseVersion: null, input: { commands: cmds(n, badLast) } }) });
    let t = Date.now(); const ok = await post('p-ok', 2000, false); const okMs = Date.now() - t;
    t = Date.now(); const bad = await post('p-bad', 2000, true); const badMs = Date.now() - t; const badBody = await bad.text();
    const accountEmptyAfterBad = Number((await sql<{ n: string }>`select count(*)::text as n from log_entries where user_id = (select id from users where auth_user_id = 'dev:p-bad')`.execute(ownerDb)).rows[0]!.n);
    // two valid imports at once from two accounts on a pool of 2, while a third account asks /api/me
    const hold = [post('h1', 2000, false), post('h2', 2000, false)];
    await new Promise(r => setTimeout(r, 400));
    t = Date.now(); const me = await fetch(`${base}/api/me`, { headers: H('victim') }); const meMs = Date.now() - t;
    await Promise.all(hold);
    console.log('P2', JSON.stringify({ import2000OkMs: okMs, okStatus: ok.status, import2000BadLastMs: badMs, badStatus: bad.status, badBody: badBody.slice(0, 160), rowsLeftAfterBad: accountEmptyAfterBad, victimMeStatus: me.status, victimMeWaitedMs: meMs, poolMax: 2 }));
    s.close(); await db.destroy();
  }, 300_000);

  it('P2b one account, concurrent imports, pool of 3', async () => {
    const good = (i: number) => ({ name: 'log-session', clientId: 'i' + i, input: { exerciseId: 'squat', entry: { d: '2026-10-05', ph: 'strength', w: 135, s: 3, r: 5 } } });
    const body = JSON.stringify({ clientId: 'import-1', baseVersion: null, input: { commands: Array.from({ length: 2000 }, (_, i) => good(i)) } });
    const db = mk(3); const { s, base } = await serve(db);
    const t0 = Date.now();
    const imps = [1, 2, 3].map(() => fetch(`${base}/api/commands/import-legacy`, { method: 'POST', headers: H('solo'), body }).then(r => ({ status: r.status, ms: Date.now() - t0 })));
    await new Promise(r => setTimeout(r, 400));
    const t = Date.now(); const me = await fetch(`${base}/api/me`, { headers: H('victim') }); const meMs = Date.now() - t;
    const out = await Promise.all(imps);
    console.log('P2b', JSON.stringify({ importResults: out, victimMeStatus: me.status, victimMeWaitedMs: meMs, poolMax: 3 }));
    s.close(); await db.destroy();
  }, 300_000);
});
