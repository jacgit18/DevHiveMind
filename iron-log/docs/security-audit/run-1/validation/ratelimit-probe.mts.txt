import type { AddressInfo } from 'node:net';
import { createApp } from '/home/jac/Videos/iron-log/server/app.ts';
const s = createApp({ trustProxy: 1 }).listen(0); await new Promise(d => s.once('listening', d));
const base = `http://127.0.0.1:${(s.address() as AddressInfo).port}`;
const hit = (ip: string) => fetch(`${base}/api/client-errors`, { method: 'POST', headers: { 'content-type': 'application/json', 'x-forwarded-for': ip }, body: '{}' }).then(r => r.status);
const ms = async (ip: string, n = 30) => { const t: number[] = []; for (let i = 0; i < n; i++) { const a = performance.now(); await fetch(`${base}/api/me`, { headers: { 'x-forwarded-for': ip } }); t.push(performance.now() - a); } t.sort((a, b) => a - b); return +t[Math.floor(n / 2)]!.toFixed(2); };
// 1. same address: limited after 20 a minute
const same: number[] = []; for (let i = 0; i < 25; i++) same.push(await hit('203.0.113.9'));
// 2. a new IPv6 address every time: never limited
const rot: number[] = []; for (let i = 0; i < 25; i++) rot.push(await hit(`2001:db8::${(i + 1).toString(16)}`));
const p50Before = await ms('203.0.113.50');
// 3. fill the map with distinct live keys, then time an ordinary request
const fill = async (n: number) => { for (let i = 0; i < n; i += 100) await Promise.all(Array.from({ length: 100 }, (_, j) => fetch(`${base}/api/me`, { headers: { 'x-forwarded-for': `2001:db8:1::${(i + j).toString(16)}` } }).then(r => r.arrayBuffer()))); };
const out: Record<string, number> = {};
for (const n of [5000, 15000, 40000]) { await fill(n === 5000 ? 5000 : n - (n === 15000 ? 5000 : 15000)); out['p50_ms_after_' + n + '_keys'] = await ms('203.0.113.51'); }
console.log(JSON.stringify({ sameAddressStatuses: { ok: same.filter(x => x !== 429).length, limited: same.filter(x => x === 429).length }, rotatingIPv6Statuses: { ok: rot.filter(x => x !== 429).length, limited: rot.filter(x => x === 429).length }, p50_ms_before: p50Before, ...out }));
s.close();
