// Unauthenticated body-size -> server memory in the better-auth mount exactly as server/app.ts L85 mounts it (no body cap before it).
// Server and client are separate processes so only the server's RSS is measured. better-auth uses its in-memory adapter (no database).
import http from 'node:http';
import { spawn } from 'node:child_process';
const SOCK = '/tmp/s2.sock';
if (process.argv[2] === 'client') {
  const mb = Number(process.argv[3]); const n = Number(process.argv[4] || 1);
  const head = '{"provider":"google","pad":"', tail = '"}';
  const chunk = Buffer.alloc(1024 * 1024, 'a');
  const total = head.length + mb * 1024 * 1024 + tail.length;
  const one = () => new Promise(res => {
    const rq = http.request({ socketPath: SOCK, method: 'POST', path: '/api/auth/sign-in/social', headers: { 'content-type': 'application/json', 'content-length': total } }, r => { r.resume(); r.on('end', () => res(r.statusCode)); });
    rq.on('error', e => res('err ' + e.code));
    rq.write(head); let i = 0;
    const pump = () => { while (i < mb) { i++; if (!rq.write(chunk)) { rq.once('drain', pump); return; } } rq.end(tail); };
    pump();
  });
  const st = await Promise.all(Array.from({ length: n }, one)); console.log(JSON.stringify(st)); process.exit(0);
}
const { default: express } = await import('express');
const { betterAuth } = await import('better-auth');
const { memoryAdapter } = await import('better-auth/adapters/memory');
const { toNodeHandler } = await import('better-auth/node');
const auth = betterAuth({ baseURL: 'http://127.0.0.1:3999', secret: 'dummy-dummy-dummy-dummy-dummy-dummy-12', database: memoryAdapter({}), socialProviders: { google: { clientId: 'a', clientSecret: 'b' } } });
const app = express();
app.all('/api/auth/*splat', toNodeHandler(auth));
const srv = app.listen(SOCK); await new Promise(r => srv.once('listening', r));
let peak = 0; setInterval(() => { peak = Math.max(peak, process.memoryUsage().rss); }, 3).unref();
const run = (mb, n) => new Promise(res => {
  global.gc?.(); const before = process.memoryUsage().rss; peak = before;
  const c = spawn(process.execPath, [new URL(import.meta.url).pathname, 'client', String(mb), String(n)], { stdio: ['ignore', 'pipe', 'inherit'] });
  let out = ''; c.stdout.on('data', d => out += d);
  c.on('close', () => setTimeout(() => { console.log(JSON.stringify({ bodyMB: mb, concurrent: n, statuses: out.trim(), rssBeforeMB: Math.round(before / 1048576), rssPeakMB: Math.round(peak / 1048576), peakDeltaMB: Math.round((peak - before) / 1048576) })); res(); }, 100));
});
console.log('node', process.version, 'idle rss MB', Math.round(process.memoryUsage().rss / 1048576));
await run(1, 1); await run(8, 1); await run(16, 1); await run(32, 1); await run(16, 3);
process.exit(0);
