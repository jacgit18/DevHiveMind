```dataviewjs
const self = dv.current();
const folder = self.file.folder;
const pages = dv.pages(`"${folder}"`).where((p) => p.file.path !== self.file.path);

dv.paragraph(`**${pages.length}** notes in this section. Auto-generated from the folder, so it updates itself.`);

dv.header(2, "Recently edited");
dv.list(pages.sort((p) => p.file.mtime, "desc").limit(5).file.link);

dv.header(2, "By subfolder");
const groups = pages
  .groupBy((p) => p.file.folder.slice(folder.length + 1) || "(top level)")
  .sort((g) => g.key, "asc");
for (const g of groups) {
  dv.header(3, `${g.key} (${g.rows.length})`);
  dv.list(g.rows.sort((p) => p.file.name, "asc").file.link);
}

const orphans = pages.where((p) => p.file.inlinks.length === 0 && p.file.outlinks.length === 0);
if (orphans.length) {
  dv.header(2, `Unlinked notes (${orphans.length})`);
  dv.list(orphans.sort((p) => p.file.name, "asc").limit(15).file.link);
}
```
